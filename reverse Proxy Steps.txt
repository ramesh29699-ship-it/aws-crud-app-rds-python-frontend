Reverse Proxy Steps:

1. create reverse proxy file in conf.d directory to host domain
sudo vi /etc/nginx/conf.d/portfolio.conf

2. Add below code to reverse proxy file:
server {
    listen 80;
    
    # Put your EC2 Public IP here for now. Change to your domain later!
    server_name YOUR_FRONTEND_PUBLIC_IP;

    # 1. Serve the Frontend HTML
    location / {
        root /usr/share/nginx/html;
        index index.html;
    }

    # 2. Reverse Proxy for the Backend API
    location /api/ {
        proxy_pass http://<YOUR_INTERNAL_ALB_DNS>;
        
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }
}

3. Restart the Nginx
sudo nginx -t
sudo systemctl restart nginx
