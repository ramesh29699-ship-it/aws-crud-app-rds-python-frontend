Reverse Proxy Steps:

1. create reverse proxy file in conf.d directory to host domain
sudo vi /etc/nginx/conf.d/portfolio.conf

2. Add below code to reverse proxy file:
server {
    listen 80;

    location /products {
        proxy_pass http://internal-BE-Alb-INTERNET-1609149333.us-east-1.elb.amazonaws.com;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    location / {
        root /usr/share/nginx/html;
        index index.html;
        try_files $uri $uri/ =404;
    }
}

3. Restart the Nginx
sudo nginx -t
sudo systemctl restart nginx
